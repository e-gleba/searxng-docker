SearXNG health endpoint for monitoring and load balancers.

📚 Quick Start Commands

# Start all services
docker compose up -d

# View logs
docker compose logs -f

# Check service status
docker compose ps

# Stop all services
docker compose down

# Rebuild MCP server
docker compose build mcp
docker compose up -d mcp

🔧 Testing Endpoints

# Test SearXNG search API
curl "http://localhost:8080/search?q=rust&format=json" | jq .

# Test MCP server tools list
curl -X POST http://localhost:8000/mcp \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":1,"method":"tools/list"}' | jq .

# Test MCP search
curl -X POST http://localhost:8000/mcp \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":2,"method":"tools/call","params":{"name":"search","arguments":{"query":"rust programming"}}}' | jq .
